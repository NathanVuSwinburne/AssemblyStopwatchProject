keypress:
      LDR R0, .LastKey  ; Load the value of the LastKey variable into register R0.
      CMP R0, #67       ; Compare the value in R0 with the ASCII code for 'C'.
      BEQ stopwatch     ; Branch to the stopwatch label if the condition is true.
endreset:
      B infiniteloop    ; Branch to the infiniteloop label.
      RFE               ; Return from exception.

      CMP R0, #69       ; Compare the value in R0 with the ASCII code for 'E'.
      BEQ End           ; Branch to the End label if the condition is true.

      LDR R0, .LastKeyAndReset ; Load the value of the LastKeyAndReset variable into register R0.

      BEQ splitorder    ; Branch to the splitorder label if the condition is true.
